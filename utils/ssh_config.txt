# ==============================================================================
# SSH Algorithm Preferences — consumed by Netmiko via ssh_config_file
# ==============================================================================
# SCOPE: Netmiko parses this file through paramiko.SSHConfig() but only acts on
# a specific subset of directives: Ciphers/MACs/KexAlgorithms/HostKeyAlgorithms
# (forwarded to Paramiko's Transport for algorithm negotiation), plus
# ProxyCommand/ProxyJump/Hostname/Port. It does NOT read or enforce
# StrictHostKeyChecking, UserKnownHostsFile, or GlobalKnownHostsFile — those
# directives are silently ignored by Netmiko regardless of what's set here.
#
# Host-key trust (new/changed device handling) is controlled entirely in
# ssh.py via ConnectHandler's ssh_strict / alt_host_keys / alt_key_file kwargs.
# See ssh.py for the actual enforcement.
# ==============================================================================

host *
    # ---------------------------------------------------------------------
    # ENCRYPTION — FIPS-approved ciphers, GCM preferred (AEAD)
    # ---------------------------------------------------------------------
    Ciphers aes256-gcm@openssh.com,aes128-gcm@openssh.com,aes256-ctr

    # ---------------------------------------------------------------------
    # MESSAGE AUTHENTICATION — -etm variants preferred (Encrypt-then-MAC)
    # ---------------------------------------------------------------------
    MACs hmac-sha2-512-etm@openssh.com,hmac-sha2-256-etm@openssh.com,hmac-sha2-512,hmac-sha2-256

    # ---------------------------------------------------------------------
    # KEY EXCHANGE
    # ---------------------------------------------------------------------
    # NOTE: mlkem768x25519 / sntrup761x25519 (post-quantum hybrid KEX) are
    # OMITTED here. They require a recent OpenSSH-compatible implementation,
    # and it has not been confirmed that Paramiko (Netmiko's underlying SSH
    # library, a separate pure-Python reimplementation of the protocol)
    # supports them. Listing algorithms Paramiko doesn't implement won't
    # cause a crash, but it won't gain you PQC protection either — verify
    # against your installed `paramiko` version before adding them back.
    KexAlgorithms curve25519-sha256,curve25519-sha256@libssh.org,ecdh-sha2-nistp521,ecdh-sha2-nistp384,ecdh-sha2-nistp256,diffie-hellman-group-exchange-sha256,diffie-hellman-group16-sha512,diffie-hellman-group18-sha512

    # ---------------------------------------------------------------------
    # HOST KEY ALGORITHMS
    # ---------------------------------------------------------------------
    HostKeyAlgorithms ecdsa-sha2-nistp521,ecdsa-sha2-nistp384,ecdsa-sha2-nistp256,rsa-sha2-512,rsa-sha2-256,ssh-ed25519

    # ---------------------------------------------------------------------
    # PUBLIC KEY ALGORITHMS (server-side preference)
    # ---------------------------------------------------------------------
    PubkeyAcceptedAlgorithms ecdsa-sha2-nistp521,ecdsa-sha2-nistp384,ecdsa-sha2-nistp256,rsa-sha2-512,rsa-sha2-256,ssh-ed25519
